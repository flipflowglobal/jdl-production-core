import sqlite3
from datetime import datetime

def record_flash_arbitrage(db_path, trade_data):
    """
    Record flash loan arbitrage execution to .db
    
    trade_data = {
        'project': 'Aureon',
        'chain': 'arbitrum',
        'asset': 'USDC',
        'amount_borrowed': 1000.0,
        'fee_paid': 5.0,
        'gross_profit': 8.5,
        'gas_cost': 0.50,
        'net_profit': 8.0,
        'tx_hash': '0x...',
        'contract_address': '0xAureonPayProcessor',
        'initiator': '0xYourWallet'
    }
    """
    
    conn = sqlite3.connect(db_path)
    c = conn.cursor()
    
    try:
        c.execute("""
            INSERT INTO flash_trades
            (project, chain, asset_borrowed, amount_borrowed, fee_paid,
             gross_profit, gas_cost, net_profit, tx_hash, 
             contract_address, initiator_address, status)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 'success')
        """, (
            trade_data['project'],
            trade_data['chain'],
            trade_data['asset'],
            trade_data['amount_borrowed'],
            trade_data['fee_paid'],
            trade_data['gross_profit'],
            trade_data['gas_cost'],
            trade_data['net_profit'],
            trade_data['tx_hash'],
            trade_data['contract_address'],
            trade_data['initiator'],
        ))
        
        conn.commit()
        print(f"✓ Recorded: {trade_data['tx_hash'][:10]}... | Profit: ${trade_data['net_profit']:.2f}")
        
    except Exception as e:
        print(f"✗ Failed to record trade: {e}")
        # Log error but don't crash the arbitrage loop
    
    finally:
        conn.close()


def record_withdrawal(db_path, withdrawal_data):
    """
    Record profit withdrawal from contract to wallet
    
    withdrawal_data = {
        'project': 'Aureon',
        'chain': 'arbitrum',
        'token': 'USDC',
        'amount': 125.50,
        'from_contract': '0xAureonPayProcessor',
        'to_address': '0xYourWallet',
        'tx_hash': '0x...',
        'gas_cost': 1.25
    }
    """
    
    conn = sqlite3.connect(db_path)
    c = conn.cursor()
    
    try:
        c.execute("""
            INSERT INTO withdrawals
            (project, chain, token, amount, from_contract, to_address, tx_hash, gas_cost, status)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'success')
        """, (
            withdrawal_data['project'],
            withdrawal_data['chain'],
            withdrawal_data['token'],
            withdrawal_data['amount'],
            withdrawal_data['from_contract'],
            withdrawal_data['to_address'],
            withdrawal_data['tx_hash'],
            withdrawal_data.get('gas_cost', 0)
        ))
        
        conn.commit()
        print(f"✓ Recorded withdrawal: {withdrawal_data['amount']} {withdrawal_data['token']}")
        
    except Exception as e:
        print(f"✗ Failed to record withdrawal: {e}")
    
    finally:
        conn.close()

TROUBLESHOOTING
─────────────────────────────────────────────────────────────────────────────

Q: "ModuleNotFoundError: No module named 'web3'"
A: Install dependencies:
   pip install web3 requests

Q: "sqlite3.OperationalError: no such table: flash_trades"
A: Schema not initialized. Run:
   bash deploy_revenue_system.sh

Q: "RPC connection timeout"
A: Check internet. Script auto-tries backup RPC. Monitor logs:
   tail -f ~/dl.2/logs/chain_monitor.log

Q: Revenue not showing in .db
A: Add record_flash_arbitrage() call after each execution

Q: "Contract address not set"
A: Edit ~/revenue_system.conf and set CONTRACT_* addresses
